def one(label, instrument=None):
    global allocation_peak, profile_summary
    with warnings.catch_warnings(record=True) as caught:
        warnings.simplefilter('always')
        if instrument=='memory': tracemalloc.start()
        if instrument=='profile': profiler=cProfile.Profile()
        wall=time.perf_counter();cpu=time.process_time()
        result=profiler.runcall(execute) if instrument=='profile' else execute()
        measured={'wall_seconds':time.perf_counter()-wall,'cpu_seconds':time.process_time()-cpu}
        if instrument=='memory':
            _,allocation_peak=tracemalloc.get_traced_memory();tracemalloc.stop()
        if instrument=='profile':
            profiler.dump_stats(str(run/'cpu.prof'))
            stats=pstats.Stats(profiler)
            profile_summary={'total_calls':stats.total_calls,'primitive_calls':stats.prim_calls,'seconds':stats.total_tt}
            with (run/'cpu.txt').open('w') as stream:
                pstats.Stats(profiler,stream=stream).sort_stats('cumulative').print_stats(60)
                pstats.Stats(profiler,stream=stream).sort_stats('tottime').print_stats(30)
    # Workloads without file output still receive an exact evidence directory.
    directory=run/f'output_{sequence:02}' if sequence else run/label
    if not sequence: directory.mkdir()
    hashes=capture_result(result,workload=args.workload,directory=directory,run=run,candidate=candidate)
    del result
    warning_records=[{'category':warning.category.__name__,'message':str(warning.message),'filename':warning.filename,'lineno':warning.lineno} for warning in caught]
    (directory/'warnings.json').write_text(json.dumps(warning_records,indent=2)+'\n')
    outputs.append({'label':label,'directory':str(directory.relative_to(run)),'hashes':hashes,'warnings':len(caught)})
    print('EXECUTED',label,measured,'output',hashes['normalized_result_sha256'],flush=True)
    return measured
